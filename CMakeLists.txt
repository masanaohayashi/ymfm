cmake_minimum_required(VERSION 3.15)
project(ymfmplus LANGUAGES CXX)
include(CTest)
add_library(ymfm STATIC
    src/ymfm_adpcm.cpp src/ymfm_misc.cpp src/ymfm_opl.cpp src/ymfm_opm.cpp
    src/ymfm_opn.cpp src/ymfm_opq.cpp src/ymfm_opz.cpp src/ymfm_pcm.cpp src/ymfm_ssg.cpp)
target_include_directories(ymfm PUBLIC src)
target_compile_features(ymfm PUBLIC cxx_std_14)
add_library(ymfm_precision INTERFACE)
target_include_directories(ymfm_precision INTERFACE src)
target_compile_features(ymfm_precision INTERFACE cxx_std_14)
if(BUILD_TESTING)
    add_executable(ymfm_buildall examples/buildall/buildall.cpp)
    target_link_libraries(ymfm_buildall PRIVATE ymfm)
    add_test(NAME original_all_cores COMMAND ymfm_buildall)
    add_executable(ymfm_precision_tests tests/precision.cpp)
    target_link_libraries(ymfm_precision_tests PRIVATE ymfm_precision ymfm)
    add_test(NAME precision COMMAND ymfm_precision_tests)
    add_executable(ymfm_precision_scalar_tests tests/precision.cpp)
    target_link_libraries(ymfm_precision_scalar_tests PRIVATE ymfm_precision ymfm)
    target_compile_definitions(ymfm_precision_scalar_tests PRIVATE YMFM_PRECISION_DISABLE_SIMD)
    add_test(NAME precision_scalar COMMAND ymfm_precision_scalar_tests)
    add_executable(ymfm_original_regression tests/original_regression.cpp)
    target_link_libraries(ymfm_original_regression PRIVATE ymfm)
    add_test(NAME original_regression COMMAND ymfm_original_regression)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND AND UNIX)
        add_test(NAME render_kernel_audit COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-render-kernel.py ${CMAKE_CXX_COMPILER})
    endif()
endif()

add_executable(ymfm_precision_example examples/precision/render.cpp)
target_link_libraries(ymfm_precision_example PRIVATE ymfm_precision)
